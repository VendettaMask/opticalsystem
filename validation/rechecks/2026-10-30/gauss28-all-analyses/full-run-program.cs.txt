using System.Diagnostics;
using System.Security.Cryptography;
using System.Text.Json;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Headless;
using Avalonia.Threading;
using OpticalSystemDesign.App.Services;
using OpticalSystemDesign.App.Theming;
using OpticalSystemDesign.Application.Services;
using OpticalSystemDesign.Tests;

namespace AllAnalysisSample;

public static class SampleApplication
{
    public static AppBuilder BuildAvaloniaApp() => AppBuilder.Configure<OpticalSystemDesign.App.App>()
        .UseSkia().UseHeadless(new AvaloniaHeadlessPlatformOptions { UseHeadlessDrawing = false });
}

internal static class Program
{
    private static readonly JsonSerializerOptions Json = new() { WriteIndented = true };

    public static async Task Main(string[] args)
    {
        var source = Path.GetFullPath(args[0]);
        var output = Path.GetFullPath(args[1]);
        Directory.CreateDirectory(output);
        var before = SHA256.HashData(await File.ReadAllBytesAsync(source));
        using var session = SafeHeadlessUnitTestSession.StartNew(typeof(SampleApplication));
        await session.Dispatch(async () =>
        {
            ThemeApplicationService.Apply(Avalonia.Application.Current!, "Light");
            UiLocalization.Apply(UiLocalization.Chinese);
            DisplayTypography.Configure(new AppSettings { Theme = "Light", Language = "zh-CN" });
            using var app = WorkbenchApplication.Create();
            await app.Documents.OpenAsync(source);
            var document = app.Documents.GetSnapshot();
            var host = new Border { Width = 1200, Height = 900, Opacity = 0 };
            var window = new Window { Width = 900, Height = 700, Content = host };
            var images = new List<object>();
            var renderFailures = new List<object>();
            window.Show(); Dispatcher.UIThread.RunJobs();
            try
            {
                using var snapshot = app.CaptureReportSnapshot();
                var keys = snapshot.Workspace.Analyses.AnalysisNames.Select(snapshot.Workspace.Analyses.CanonicalKey)
                    .Where(MarkdownReportExporter.IsSupportedAnalysis).Distinct().ToArray();
                await File.WriteAllTextAsync(Path.Combine(output, "selection.json"), JsonSerializer.Serialize(keys.Select(key =>
                    new { Key = key, Title = WorkbenchAnalysisCatalog.DisplayName(key),
                        Parameters = snapshot.Workspace.Analyses.MergeSettings(key, null) }), Json));
                Console.WriteLine($"SELECTED {keys.Length} analyses + 2 layouts");
                var renderer = new ReportImageRenderer(host);
                var clock = Stopwatch.StartNew();
                using var timeout = new CancellationTokenSource(TimeSpan.FromMinutes(30));
                var result = await Task.Run(() => MarkdownReportExporter.ExportAsync(snapshot,
                    new MarkdownReportSettings { Format = DesignReportFormat.Pdf, AnalysisKeys = [.. keys] },
                    Path.Combine(output, "Pdf"), async (request, directory, stem, token) =>
                    {
                        try
                        {
                            var files = await renderer.WriteAsync(request, directory, stem, token);
                            foreach (var file in files)
                            {
                                var bytes = await File.ReadAllBytesAsync(Path.Combine(directory, file), token);
                                using var bitmap = SkiaSharp.SKBitmap.Decode(bytes);
                                if (bitmap.Width != 2404 || bitmap.Height != 1804)
                                    throw new InvalidOperationException($"Unexpected source size {bitmap.Width}x{bitmap.Height}");
                                var corner = bitmap.GetPixel(0, 0);
                                for (var x = 0; x < bitmap.Width; x++)
                                    if (bitmap.GetPixel(x, 0) != corner || bitmap.GetPixel(x, bitmap.Height - 1) != corner)
                                        throw new InvalidOperationException("Incomplete top/bottom border");
                                for (var y = 0; y < bitmap.Height; y++)
                                    if (bitmap.GetPixel(0, y) != corner || bitmap.GetPixel(bitmap.Width - 1, y) != corner)
                                        throw new InvalidOperationException("Incomplete side border");
                                var capture = Path.Combine(output, "source-images");
                                Directory.CreateDirectory(capture);
                                await File.WriteAllBytesAsync(Path.Combine(capture, file), bytes, token);
                                images.Add(new { Title = request.Title, File = file, bitmap.Width, bitmap.Height,
                                    Sha256 = Convert.ToHexString(SHA256.HashData(bytes)), Frame = "complete" });
                            }
                            return files;
                        }
                        catch (Exception error) when (error is not OperationCanceledException)
                        {
                            renderFailures.Add(new { request.Title, Error = error.Message });
                            Console.WriteLine($"RENDER FAILED {request.Title}: {error.Message}");
                            throw;
                        }
                    }, new Progress<ReportExportProgress>(value => Console.WriteLine(
                        $"[{clock.Elapsed:hh\\:mm\\:ss}] {value.Completed}/{value.Total} {value.Item}")), timeout.Token));
                var after = SHA256.HashData(await File.ReadAllBytesAsync(source));
                if (!before.SequenceEqual(after)) throw new InvalidOperationException("The source lens changed.");
                if (result.Succeeded + result.Failed != keys.Length + 2)
                    throw new InvalidOperationException("Not every selected item has an outcome.");
                await File.WriteAllTextAsync(Path.Combine(output, "generation.json"), JsonSerializer.Serialize(new
                {
                    Source = source, Sha256 = Convert.ToHexString(before).ToLowerInvariant(),
                    Snapshot = document, snapshot.CapturedAt, SourceUnchanged = true,
                    SelectedCount = keys.Length, Analyses = keys, ElapsedSeconds = clock.Elapsed.TotalSeconds,
                    Result = result, Images = images, RenderFailures = renderFailures
                }, Json));
                Console.WriteLine($"COMPLETE {result.Succeeded} succeeded / {result.Failed} failed; {images.Count} images; {result.ReportPath}");
            }
            finally { window.Close(); }
        }, CancellationToken.None);
    }
}
